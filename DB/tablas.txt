-- ============================================================================
-- 2. MÓDULO DE CATÁLOGO
-- ============================================================================

CREATE TABLE categorias (
    id          NUMBER GENERATED ALWAYS AS IDENTITY,
    nombre      VARCHAR2(100) NOT NULL,
    descripcion VARCHAR2(255),
    estado      NUMBER(1) DEFAULT 1 NOT NULL, -- 1 = Activo, 0 = Inactivo
    
    CONSTRAINT pk_categorias PRIMARY KEY (id),
    CONSTRAINT chk_cat_estado CHECK (estado IN (0, 1))
);

CREATE TABLE productos (
    id              NUMBER GENERATED ALWAYS AS IDENTITY,
    categoria_id    NUMBER NOT NULL,
    codigo_barras   VARCHAR2(50),
    nombre          VARCHAR2(150) NOT NULL,
    descripcion     VARCHAR2(500),
    precio_venta    NUMBER(10,2) NOT NULL,
    stock           NUMBER(10) DEFAULT 0 NOT NULL,
    imagen_url      VARCHAR2(255),
    estado          NUMBER(1) DEFAULT 1 NOT NULL,
    
    CONSTRAINT pk_productos PRIMARY KEY (id),
    CONSTRAINT fk_prod_categoria FOREIGN KEY (categoria_id) REFERENCES categorias(id),
    CONSTRAINT uq_prod_codigo UNIQUE (codigo_barras),
    CONSTRAINT chk_prod_precio CHECK (precio_venta >= 0),
    CONSTRAINT chk_prod_stock CHECK (stock >= 0),
    CONSTRAINT chk_prod_estado CHECK (estado IN (0, 1))
);


-- ============================================================================
-- 3. MÓDULO DE ACTORES (USUARIOS Y CLIENTES)
-- ============================================================================

CREATE TABLE usuarios (
    id       NUMBER GENERATED ALWAYS AS IDENTITY,
    nombre   VARCHAR2(150) NOT NULL,
    email    VARCHAR2(100) NOT NULL,
    password VARCHAR2(255) NOT NULL,
    rol      VARCHAR2(30) NOT NULL,
    estado   NUMBER(1) DEFAULT 1 NOT NULL,
    
    CONSTRAINT pk_usuarios PRIMARY KEY (id),
    CONSTRAINT uq_user_email UNIQUE (email),
    CONSTRAINT chk_user_estado CHECK (estado IN (0, 1))
);

CREATE TABLE clientes (
    id          NUMBER GENERATED ALWAYS AS IDENTITY,
    nit_o_dpi   VARCHAR2(20), -- Adaptable según el documento de identificación
    nombre      VARCHAR2(150) NOT NULL,
    telefono    VARCHAR2(20),
    direccion   VARCHAR2(255),
    
    CONSTRAINT pk_clientes PRIMARY KEY (id),
    CONSTRAINT uq_cliente_doc UNIQUE (nit_o_dpi)
);


-- ============================================================================
-- 4. MÓDULO DE TRANSACCIONES (VENTAS)
-- ============================================================================

CREATE TABLE ventas (
    id          NUMBER GENERATED ALWAYS AS IDENTITY,
    cliente_id  NUMBER NOT NULL,
    usuario_id  NUMBER NOT NULL,
    fecha_venta TIMESTAMP DEFAULT CURRENT_TIMESTAMP NOT NULL,
    total       NUMBER(10,2) NOT NULL,
    tipo_pago   VARCHAR2(50) NOT NULL,
    
    CONSTRAINT pk_ventas PRIMARY KEY (id),
    CONSTRAINT fk_ventas_cliente FOREIGN KEY (cliente_id) REFERENCES clientes(id),
    CONSTRAINT fk_ventas_usuario FOREIGN KEY (usuario_id) REFERENCES usuarios(id),
    CONSTRAINT chk_ventas_total CHECK (total >= 0)
);

CREATE TABLE detalle_ventas (
    id              NUMBER GENERATED ALWAYS AS IDENTITY,
    venta_id        NUMBER NOT NULL,
    producto_id     NUMBER NOT NULL,
    cantidad        NUMBER(10) NOT NULL,
    precio_unitario NUMBER(10,2) NOT NULL,
    
    CONSTRAINT pk_detalle_ventas PRIMARY KEY (id),
    -- ON DELETE CASCADE elimina los detalles si por alguna razón se borra la cabecera
    CONSTRAINT fk_det_venta FOREIGN KEY (venta_id) REFERENCES ventas(id) ON DELETE CASCADE,
    CONSTRAINT fk_det_producto FOREIGN KEY (producto_id) REFERENCES productos(id),
    CONSTRAINT chk_det_cantidad CHECK (cantidad > 0),
    CONSTRAINT chk_det_precio CHECK (precio_unitario >= 0)
);

SELECT 
    v.id AS no_factura,
    TO_CHAR(v.fecha_venta, 'DD/MM/YYYY HH24:MI:SS') AS fecha,
    c.nombre AS cliente,
    u.nombre AS atendido_por,
    p.nombre AS producto,
    dv.cantidad,
    dv.precio_unitario,
    (dv.cantidad * dv.precio_unitario) AS subtotal,
    v.total AS gran_total,
    v.tipo_pago
FROM ventas v
JOIN clientes c ON v.cliente_id = c.id
JOIN usuarios u ON v.usuario_id = u.id
JOIN detalle_ventas dv ON v.id = dv.venta_id
JOIN productos p ON dv.producto_id = p.id
ORDER BY v.id DESC;
